cmake_minimum_required(VERSION 3.16)
project(conpr_file_player)
get_filename_component(PLAYER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
get_filename_component(PLAYER_ROOT "${PLAYER_SOURCE}/../.." ABSOLUTE)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
ament_python_install_package(conpr_player PACKAGE_DIR "${PLAYER_ROOT}/conpr_player")
set(PLAYER_PYTHON "${Python3_EXECUTABLE}")
configure_file("${PLAYER_ROOT}/scripts/player.in" "${CMAKE_CURRENT_BINARY_DIR}/player" @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/player" DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch)
install(DIRECTORY "${PLAYER_ROOT}/config/" DESTINATION share/${PROJECT_NAME}/config FILES_MATCHING PATTERN "*.rviz")
install(FILES "${PLAYER_ROOT}/LICENSE" DESTINATION share/${PROJECT_NAME})
ament_package()
