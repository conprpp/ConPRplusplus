cmake_minimum_required(VERSION 3.16)
project(conpr_file_player)
get_filename_component(PLAYER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
get_filename_component(PLAYER_ROOT "${PLAYER_SOURCE}/../.." ABSOLUTE)
find_package(catkin REQUIRED)
catkin_python_setup()
catkin_package()
set(PLAYER_PYTHON "${PYTHON_EXECUTABLE}")
configure_file("${PLAYER_ROOT}/scripts/player.in" "${CMAKE_CURRENT_BINARY_DIR}/player" @ONLY)
catkin_install_python(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/player" DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/launch)
install(DIRECTORY "${PLAYER_ROOT}/config/" DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/config FILES_MATCHING PATTERN "*.rviz")
install(FILES "${PLAYER_ROOT}/LICENSE" DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
